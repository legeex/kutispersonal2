function esc(text) {
  return String(text == null ? "" : text)
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/\n/g, "<br>");
}

function labelOf(key) {
  return LABEL[key] || key;
}

function rowsFrom(payload) {
  var rows = [];
  var used = {};
  var i, key, val, lab;

  for (i = 0; i < PORADI.length; i++) {
    key = PORADI[i];
    val = payload[key];
    if (!val || String(val).trim() === "") continue;
    lab = labelOf(key);
    if (used[lab]) continue;
    rows.push({ label: lab, value: String(val).trim(), key: key });
    used[lab] = true;
  }

  for (key in payload) {
    if (!payload.hasOwnProperty(key)) continue;
    if (key === "subject" || used[labelOf(key)]) continue;
    val = payload[key];
    if (val === "" || val == null) continue;
    lab = labelOf(key);
    rows.push({ label: lab, value: String(val).trim(), key: key });
    used[lab] = true;
  }

  return rows;
}

function isCompany(typ) {
  return String(typ || "").toLowerCase().indexOf("firma") !== -1;
}

function nowCs() {
  return new Date().toLocaleString("cs-CZ", {
    timeZone: "Europe/Prague",
    day: "numeric",
    month: "numeric",
    year: "numeric",
    hour: "2-digit",
    minute: "2-digit",
  });
}

function valueHtml(row) {
  var v = esc(row.value);
  var k = String(row.key || row.label).toLowerCase();
  if (k === "email" || row.label === "E-mail") {
    return (
      '<a href="mailto:' +
      esc(row.value) +
      '" style="color:#c4121a;font-weight:600;text-decoration:none;">' +
      v +
      "</a>"
    );
  }
  if (k === "phone" || row.label === "Telefon") {
    var tel = String(row.value).replace(/\s+/g, "");
    return (
      '<a href="tel:' +
      esc(tel) +
      '" style="color:#111;font-weight:600;text-decoration:none;">' +
      v +
      "</a>"
    );
  }
  return v;
}

function setup() {
  var sh = sheet_();
  Logger.log("Tabulka s poptávkami: " + sh.getParent().getUrl());
  Logger.log("Zbývající denní kvóta e-mailů: " + MailApp.getRemainingDailyQuota());
}

function sheet_() {
  var props = PropertiesService.getScriptProperties();
  var id = props.getProperty("SHEET_ID");
  var ss = null;
  if (id) {
    try { ss = SpreadsheetApp.openById(id); } catch (err) { ss = null; }
  }
  if (!ss) {
    ss = SpreadsheetApp.create(NAZEV_TABULKY);
    props.setProperty("SHEET_ID", ss.getId());
    var first = ss.getSheets()[0];
    first.setName("Poptávky");
    first.appendRow(["Čas", "Stav", "Typ", "Jazyk", "Stránka", "Údaje"]);
    first.setFrozenRows(1);
  }
  return ss.getSheetByName("Poptávky") || ss.getSheets()[0];
}

function cell_(v) {
  v = String(v == null ? "" : v);
  return /^[=+\-@]/.test(v) ? "'" + v : v; // žádné vzorce z formuláře
}

function log_(stav, payload) {
  try {
    var rows = rowsFrom(payload);
    var text = rows.map(function (r) { return r.label + ": " + r.value; }).join("\n");
    sheet_().appendRow([
      new Date(), stav, cell_(payload["Typ formuláře"]), cell_(payload["Jazyk webu"]),
      cell_(payload["Stránka"]), cell_(text),
    ]);
  } catch (err) {}
}

function clean_(raw) {
  if (!raw || typeof raw !== "object" || Array.isArray(raw)) return null;
  var out = {};
  var n = 0;
  for (var key in raw) {
    if (!raw.hasOwnProperty(key)) continue;
    if (++n > 40) break;
    var val = raw[key];
    if (val == null || typeof val === "object") continue;
    out[String(key).slice(0, 60)] = String(val).slice(0, MAX_DELKA_POLE);
  }
  return out;
}

function isValid_(p) {
  if (p.website || p._gotcha) return false; // skryté pole vyplnil robot
  var typ = String(p["Typ formuláře"] || "");
  if (typ !== "Firma" && typ !== "Pracovník / partner") return false;
  for (var key in p) {
    if (/telefon|e-mail|email|phone|kontakt|contact/i.test(key) && String(p[key]).trim().length >= 5) return true;
  }
  return false;
}

function fingerprint_(p) {
  var keys = Object.keys(p).filter(function (k) { return k !== "Stránka" && k !== "subject"; }).sort();
  var s = keys.map(function (k) { return k + "=" + String(p[k]).trim().toLowerCase(); }).join("|");
  var d = Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, s, Utilities.Charset.UTF_8);
  return "dup_" + Utilities.base64EncodeWebSafe(d);
}

function bump_(prefix) {
  // prefix "r" = všechny požadavky (ochrana proti útoku), "m" = odeslané e-maily
  var now = new Date();
  var hourKey = prefix + "h_" + Utilities.formatDate(now, "Europe/Prague", "yyyyMMddHH");
  var dayKey = prefix + "d_" + Utilities.formatDate(now, "Europe/Prague", "yyyyMMdd");
  var props = PropertiesService.getScriptProperties();
  var lock = LockService.getScriptLock();
  lock.tryLock(5000);
  try {
    var all = props.getProperties();
    for (var k in all) { // úklid starých počítadel
      if (k.indexOf(prefix + "h_") === 0 && k !== hourKey) props.deleteProperty(k);
      if (k.indexOf(prefix + "d_") === 0 && k !== dayKey) props.deleteProperty(k);
    }
    var h = Number(all[hourKey] || 0) + 1;
    var d = Number(all[dayKey] || 0) + 1;
    props.setProperty(hourKey, String(h));
    props.setProperty(dayKey, String(d));
    return { hour: h, day: d };
  } finally {
    lock.releaseLock();
  }
}

function mailCount_() {
  var now = new Date();
  var props = PropertiesService.getScriptProperties();
  return {
    hour: Number(props.getProperty("mh_" + Utilities.formatDate(now, "Europe/Prague", "yyyyMMddHH")) || 0),
    day: Number(props.getProperty("md_" + Utilities.formatDate(now, "Europe/Prague", "yyyyMMdd")) || 0),
  };
}

function alertOnce_() {
  var props = PropertiesService.getScriptProperties();
  var key = "alert_" + Utilities.formatDate(new Date(), "Europe/Prague", "yyyyMMdd");
  if (props.getProperty(key)) return;
  var all = props.getProperties();
  for (var k in all) if (k.indexOf("alert_") === 0) props.deleteProperty(k);
  props.setProperty(key, "1");
  try {
    MailApp.sendEmail({
      to: EMAIL_PRIJEMCE,
      subject: "⚠️ Kutis Personal – dosažen limit e-mailů z webu",
      body: "Z webu dnes přišlo neobvykle mnoho zpráv. Další se už neposílají e-mailem, ale ukládají se do tabulky:\n\n" +
        sheet_().getParent().getUrl() + "\n\nZkontroluj ji prosím – mohou tam být i skutečné poptávky.",
    });
  } catch (err) {}
}

function ok_() {
  return ContentService.createTextOutput(JSON.stringify({ success: true }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var body = e && e.postData && e.postData.contents;
    if (!body || body.length > MAX_VELIKOST) return ok_();

    var r = bump_("r");
    if (r.hour > STOP_ZA_HODINU) return ok_(); // útok – nic neukládat

    var payload = clean_(JSON.parse(body));
    if (!payload) return ok_();
    if (!isValid_(payload)) {
      log_("odmítnuto (spam / neplatné)", payload);
      return ok_();
    }

    var cache = CacheService.getScriptCache();
    var fp = fingerprint_(payload);
    if (cache.get(fp)) {
      log_("duplicita – neodesláno", payload);
      return ok_();
    }
    cache.put(fp, "1", 21600); // 6 hodin

    var m = mailCount_();
    if (m.hour >= LIMIT_ZA_HODINU || m.day >= LIMIT_ZA_DEN || MailApp.getRemainingDailyQuota() < MIN_KVOTA) {
      log_("jen uloženo (limit)", payload);
      alertOnce_();
      return ok_();
    }

    var copy = JSON.parse(JSON.stringify(payload));
    copy.subject = String(copy.subject || "").slice(0, 150);
    var mail = buildEmail(copy);
    MailApp.sendEmail({
      to: EMAIL_PRIJEMCE,
      subject: mail.subject,
      htmlBody: mail.html,
      body: mail.plain,
    });
    bump_("m");
    log_("odesláno", payload);
    return ok_();
  } catch (err) {
    try { log_("chyba: " + String(err).slice(0, 200), clean_(JSON.parse(e.postData.contents)) || {}); } catch (x) {}
    return ContentService.createTextOutput(JSON.stringify({ success: false }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
