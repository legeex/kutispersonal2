function esc(text) {
  return String(text == null ? "" : text)
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/\n/g, "<br>");
}

function labelOf(key) {
  return LABEL[key] || key;
}

function rowsFrom(payload) {
  var rows = [];
  var used = {};
  var i, key, val, lab;

  for (i = 0; i < PORADI.length; i++) {
    key = PORADI[i];
    val = payload[key];
    if (!val || String(val).trim() === "") continue;
    lab = labelOf(key);
    if (used[lab]) continue;
    rows.push({ label: lab, value: String(val).trim(), key: key });
    used[lab] = true;
  }

  for (key in payload) {
    if (!payload.hasOwnProperty(key)) continue;
    if (key === "subject" || used[labelOf(key)]) continue;
    val = payload[key];
    if (val === "" || val == null) continue;
    lab = labelOf(key);
    rows.push({ label: lab, value: String(val).trim(), key: key });
    used[lab] = true;
  }

  return rows;
}

function isCompany(typ) {
  return String(typ || "").toLowerCase().indexOf("firma") !== -1;
}

function nowCs() {
  return new Date().toLocaleString("cs-CZ", {
    timeZone: "Europe/Prague",
    day: "numeric",
    month: "numeric",
    year: "numeric",
    hour: "2-digit",
    minute: "2-digit",
  });
}

function valueHtml(row) {
  var v = esc(row.value);
  var k = String(row.key || row.label).toLowerCase();
  if (k === "email" || row.label === "E-mail") {
    return (
      '<a href="mailto:' +
      esc(row.value) +
      '" style="color:#c4121a;font-weight:600;text-decoration:none;">' +
      v +
      "</a>"
    );
  }
  if (k === "phone" || row.label === "Telefon") {
    var tel = String(row.value).replace(/\s+/g, "");
    return (
      '<a href="tel:' +
      esc(tel) +
      '" style="color:#111;font-weight:600;text-decoration:none;">' +
      v +
      "</a>"
    );
  }
  return v;
}

function buildEmail(payload) {
  var subject = payload.subject || "Nová zpráva – Kutis Personal";
  delete payload.subject;

  var typ = payload["Typ formuláře"] || "";
  var firm = isCompany(typ);
  var badge = firm ? "Firemní poptávka" : "Registrace";
  var rows = rowsFrom(payload);
  var cas = nowCs();

  var table = "";
  for (var i = 0; i < rows.length; i++) {
    var r = rows[i];
    var last = i === rows.length - 1;
    table +=
      "<tr>" +
      '<td style="padding:12px 0;width:34%;vertical-align:top;font:600 11px/1.4 Helvetica,Arial,sans-serif;color:#888;text-transform:uppercase;letter-spacing:.06em;border-bottom:' +
      (last ? "0" : "1px solid #eee") +
      ';">' +
      esc(r.label) +
      "</td>" +
      '<td style="padding:12px 0 12px 16px;vertical-align:top;font:400 15px/1.55 Helvetica,Arial,sans-serif;color:#111;border-bottom:' +
      (last ? "0" : "1px solid #eee") +
      ';">' +
      valueHtml(r) +
      "</td>" +
      "</tr>";
  }

  var html =
    '<!DOCTYPE html><html><head><meta charset="UTF-8"><meta name="viewport" content="width=device-width"></head>' +
    '<body style="margin:0;padding:0;background:#f3f3f3;font-family:Helvetica,Arial,sans-serif;">' +
    '<table role="presentation" width="100%" cellspacing="0" cellpadding="0" style="background:#f3f3f3;padding:28px 12px;">' +
    '<tr><td align="center">' +
    '<table role="presentation" width="560" cellspacing="0" cellpadding="0" style="max-width:560px;width:100%;background:#fff;border-radius:4px;overflow:hidden;border:1px solid #e6e6e6;">' +
    /* top bar */
    '<tr><td style="height:4px;background:#c4121a;font-size:0;line-height:0;">&nbsp;</td></tr>' +
    /* header */
    '<tr><td style="padding:28px 32px 20px;background:#111;">' +
    '<div style="font:700 11px/1 Helvetica,Arial,sans-serif;color:#c4121a;letter-spacing:.16em;text-transform:uppercase;">Kutis Personal</div>' +
    '<div style="margin-top:10px;font:700 22px/1.25 Helvetica,Arial,sans-serif;color:#fff;">Nová zpráva z webu</div>' +
    '<div style="margin-top:6px;font:400 13px/1.4 Helvetica,Arial,sans-serif;color:#aaa;">Václav Kutiš</div>' +
    "</td></tr>" +
    /* badge */
    '<tr><td style="padding:22px 32px 0;">' +
    '<span style="display:inline-block;padding:7px 12px;background:' +
    (firm ? "#111" : "#c4121a") +
    ";color:#fff;font:700 10px/1 Helvetica,Arial,sans-serif;letter-spacing:.1em;text-transform:uppercase;border-radius:2px;\">" +
    esc(badge) +
    "</span></td></tr>" +
    /* content */
    '<tr><td style="padding:18px 32px 28px;">' +
    '<table role="presentation" width="100%" cellspacing="0" cellpadding="0">' +
    table +
    "</table></td></tr>" +
    /* footer */
    '<tr><td style="padding:16px 32px;background:#fafafa;border-top:1px solid #eee;">' +
    '<p style="margin:0;font:400 12px/1.5 Helvetica,Arial,sans-serif;color:#999;">Odesláno ' +
    esc(cas) +
    " · kutispersonal.com</p>" +
    "</td></tr>" +
    "</table></td></tr></table></body></html>";

  var plain = ["NOVÁ ZPRÁVA – KUTIS PERSONAL", ""];
  for (i = 0; i < rows.length; i++) {
    plain.push(rows[i].label + ": " + rows[i].value);
    plain.push("");
  }
  plain.push("Odesláno: " + cas);

  return { subject: subject, html: html, plain: plain.join("\n") };
}

function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    var mail = buildEmail(payload);

    MailApp.sendEmail({
      to: EMAIL_PRIJEMCE,
      subject: mail.subject,
      htmlBody: mail.html,
      body: mail.plain,
    });

    return ContentService.createTextOutput(
      JSON.stringify({ success: true })
    ).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(
      JSON.stringify({ success: false, message: String(err) })
    ).setMimeType(ContentService.MimeType.JSON);
  }
}
