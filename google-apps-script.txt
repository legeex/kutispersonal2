var EMAIL_PRIJEMCE = "vaclav.kutis99@gmail.com";

var PORADI_POLI = [
  "Typ formuláře",
  "Jazyk webu",
  "Jméno a příjmení",
  "Název firmy",
  "Kontaktní osoba",
  "E-mail",
  "Telefon",
  "Profese",
  "Praxe",
  "Jazykové znalosti",
  "Země",
  "Lokalita projektu",
  "Termín / dostupnost",
  "Forma spolupráce",
  "Poznámka",
];

function escapeHtml(text) {
  return String(text)
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/\n/g, "<br>");
}

function ziskatRadky(payload) {
  var radky = [];
  var pouzite = {};

  for (var i = 0; i < PORADI_POLI.length; i++) {
    var klic = PORADI_POLI[i];
    if (payload[klic] && String(payload[klic]).trim() !== "") {
      radky.push({ label: klic, value: String(payload[klic]).trim() });
      pouzite[klic] = true;
    }
  }

  for (var key in payload) {
    if (payload.hasOwnProperty(key) && !pouzite[key] && payload[key] !== "") {
      radky.push({ label: key, value: String(payload[key]).trim() });
    }
  }

  return radky;
}

function badgeBarva(typFormulare) {
  var t = String(typFormulare || "").toLowerCase();
  if (t.indexOf("firma") !== -1) {
    return { bg: "#1a3352", text: "#e8c96a", label: "Poptávka od firmy" };
  }
  return { bg: "#1a3352", text: "#e8c96a", label: "Žádost uchazeče" };
}

function vytvorHtmlEmail(payload) {
  var subject = payload.subject || "Nová zpráva z webu – Václav Kutiš";
  delete payload.subject;

  var radky = ziskatRadky(payload);
  var typ = payload["Typ formuláře"] || "";
  var badge = badgeBarva(typ);
  var cas = new Date().toLocaleString("cs-CZ", {
    timeZone: "Europe/Prague",
    day: "numeric",
    month: "numeric",
    year: "numeric",
    hour: "2-digit",
    minute: "2-digit",
  });

  var rowsHtml = "";
  for (var j = 0; j < radky.length; j++) {
    var r = radky[j];
    var isEmail = r.label === "E-mail";
    var valueHtml = escapeHtml(r.value);
    if (isEmail) {
      valueHtml =
        '<a href="mailto:' +
        escapeHtml(r.value) +
        '" style="color:#0c1a2e;font-weight:600;text-decoration:none;">' +
        valueHtml +
        "</a>";
    }

    rowsHtml +=
      '<tr>' +
      '<td style="padding:14px 16px;border-bottom:1px solid #eef0f4;width:38%;vertical-align:top;font-size:12px;font-weight:700;color:#8b95a5;text-transform:uppercase;letter-spacing:0.06em;">' +
      escapeHtml(r.label) +
      "</td>" +
      '<td style="padding:14px 16px;border-bottom:1px solid #eef0f4;font-size:15px;color:#2d3544;line-height:1.5;">' +
      valueHtml +
      "</td>" +
      "</tr>";
  }

  var html =
    '<!DOCTYPE html><html><head><meta charset="UTF-8"></head>' +
    '<body style="margin:0;padding:0;background-color:#eef0f4;font-family:Helvetica,Arial,sans-serif;">' +
    '<table role="presentation" width="100%" cellspacing="0" cellpadding="0" style="background-color:#eef0f4;padding:32px 16px;">' +
    "<tr><td align=\"center\">" +
    '<table role="presentation" width="600" cellspacing="0" cellpadding="0" style="max-width:600px;width:100%;background:#ffffff;border-radius:12px;overflow:hidden;box-shadow:0 8px 32px rgba(12,26,46,0.12);">' +
    /* Header */
    '<tr><td style="background:linear-gradient(135deg,#0c1a2e 0%,#1a3352 100%);padding:28px 32px;">' +
    '<table role="presentation" width="100%" cellspacing="0" cellpadding="0"><tr>' +
    '<td style="font-size:11px;font-weight:700;color:#e8c96a;letter-spacing:0.14em;text-transform:uppercase;">DACH · Propojení</td>' +
    "</tr><tr>" +
    '<td style="padding-top:10px;font-size:24px;font-weight:400;color:#ffffff;font-family:Georgia,\'Times New Roman\',serif;">Nová zpráva z webu</td>' +
    "</tr><tr>" +
    '<td style="padding-top:6px;font-size:14px;color:rgba(255,255,255,0.75);">Václav Kutiš</td>' +
    "</tr></table></td></tr>" +
    /* Badge */
    '<tr><td style="padding:24px 32px 8px;">' +
    '<span style="display:inline-block;padding:6px 14px;background:' +
    badge.bg +
    ";color:" +
    badge.text +
    ';font-size:11px;font-weight:700;letter-spacing:0.08em;text-transform:uppercase;border-radius:100px;border:1px solid rgba(201,162,39,0.35);">' +
    badge.label +
    "</span></td></tr>" +
    /* Table */
    '<tr><td style="padding:8px 16px 24px;">' +
    '<table role="presentation" width="100%" cellspacing="0" cellpadding="0" style="border:1px solid #eef0f4;border-radius:8px;overflow:hidden;">' +
    rowsHtml +
    "</table></td></tr>" +
    /* Footer */
    '<tr><td style="background:#f7f8fa;padding:18px 32px;border-top:1px solid #eef0f4;">' +
    '<p style="margin:0;font-size:12px;color:#8b95a5;line-height:1.5;">Odesláno ' +
    escapeHtml(cas) +
    " · web Václav Kutiš</p>" +
    "</td></tr>" +
    "</table></td></tr></table></body></html>";

  return { subject: subject, html: html, plain: vytvorPlainEmail(radky, cas, subject) };
}

function vytvorPlainEmail(radky, cas, subject) {
  var lines = ["NOVÁ ZPRÁVA Z WEBU – VÁCLAV KUTIŠ", ""];
  for (var i = 0; i < radky.length; i++) {
    lines.push(radky[i].label + ": " + radky[i].value);
    lines.push("");
  }
  lines.push("Odesláno: " + cas);
  return lines.join("\n");
}

function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    var email = vytvorHtmlEmail(payload);

    MailApp.sendEmail({
      to: EMAIL_PRIJEMCE,
      subject: email.subject,
      htmlBody: email.html,
      body: email.plain,
    });

    return ContentService.createTextOutput(
      JSON.stringify({ success: true })
    ).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(
      JSON.stringify({ success: false, message: err.toString() })
    ).setMimeType(ContentService.MimeType.JSON);
  }
}
